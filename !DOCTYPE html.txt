function roomOf(r){
  const b=val(r,'Building Code'), d=val(r,'Issue Description'), i=d.indexOf(b);

}

  ins('stateChart',[`<b>${res}</b> of ${t} records resolved (${Math.round(res/t*100)}%)`,open.length?`${open.length} still open: ${rca} Pending RCA, ${fip} Fix in Progress`:'No open records']);
